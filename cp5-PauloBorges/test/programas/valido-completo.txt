const int LIMITE = 10;
int pot(int base, int exp) {
    return base ** exp;
}
void mostra(string msg) {
    print(msg);
    return;
}
void main() {
    int n = 2;
    double d = 3.5;
    bool ok = true;
    char c = 'a';
    string s = "ola";
    int p = pot(n, 3);
    double r = -d + n * 2.0 / 4;
    int resto = p % 3;
    read(n);
    read(d);
    read(c);
    if (n > 0 && !ok) {
        mostra(s + "!");
    } elif (c <= 'z' || n == d) {
        while (n > 0 && n < LIMITE) {
            n = n - 1;
        }
    } else {
        print(resto);
    }
}
