void main() {
    int x = 0;
    read(x);
    if ((x > 20)) {
        print("Acima do intervalo");
    } elif (((x > 10) && (x <= 20))) {
        print("Dentro do intervalo");
    } else {
        print("Abaixo do intervalo");
    }
    bool executando = true;
    while (executando) {
        x = (x - 1);
        if ((x == 0)) {
            executando = false;
        }
    }
}
